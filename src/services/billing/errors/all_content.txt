import { BillingErrorCode } from "./BillingErrorCode";

export interface BillingErrorOptions {
  code: BillingErrorCode;
  message: string;
  statusCode?: number;
  metadata?: Record<string, unknown>;
  cause?: Error;
}

export class BillingError extends Error {
  readonly code: BillingErrorCode;
  readonly statusCode: number;
  readonly metadata?: Record<string, unknown>;

  constructor(options: BillingErrorOptions) {
    super(options.message, {
      cause: options.cause,
    });

    this.name = new.target.name;

    this.code = options.code;
    this.statusCode = options.statusCode ?? 400;
    this.metadata = options.metadata ?? {};

    Error.captureStackTrace?.(this, this.constructor);
  }

  serialize() {
    return {
      error: this.name,
      code: this.code,
      message: this.message,
      statusCode: this.statusCode,
      metadata: this.metadata,
    };
  }

   toJSON() {
    return this.serialize();
  }
}

import { BillingError } from "./BillingError";
import { BillingErrorCode } from "./BillingErrorCode";

export class QuotaExceededError extends BillingError {
  constructor(
    quotaKey: string,
    limit: number,
    usage: number,
    requested: number = 1
  ) {
    super({
      code: BillingErrorCode.QUOTA_EXCEEDED,
      statusCode: 403,
      message: `Quota '${quotaKey}' has been exceeded.`,
      metadata: {
        quotaKey,
        limit,
        usage,
        requested,
        remaining: Math.max(limit - usage, 0),
      },
    });
  }
}
